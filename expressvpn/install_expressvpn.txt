# unpacking *.run and installing expressvpn manually (x64 version)

(cd Завантажене)
./expressvpn-linux-universal-14.2.1.13658_release.run --noexec --keep --target /tmp/expressvpn

# Creating the necessary groups (skip if already installed/updating)
sudo groupadd expressvpn
sudo groupadd expressvpnhnsd

# Creating a directory
sudo mkdir -p /opt/expressvpn/{bin,etc,var,share,lib,plugins,qml}

# Stop the running daemon before overwriting files (update only)
sudo systemctl stop expressvpn

# Copying files from the unpacked archive
sudo cp -r /tmp/expressvpn/x64/expressvpnfiles/* /opt/expressvpn/
sudo cp /tmp/expressvpn/x64/installfiles/*.sh /opt/expressvpn/bin/
sudo chmod +x /opt/expressvpn/bin/*.sh

# Setting permissions for unbound
sudo setcap 'cap_net_bind_service=+ep' /opt/expressvpn/bin/expressvpn-unbound

# Copying the icon and .desktop file
sudo cp /tmp/expressvpn/x64/installfiles/expressvpn.desktop /home/<username>/.local/share/applications/expressvpn.desktop
sudo cp /tmp/expressvpn/x64/installfiles/app-icon.png /home/<username>/.local/share/icons/hicolor/256x256/apps/expressvpn.png

# Configuring NetworkManager for WireGuard
sudo mkdir -p /etc/NetworkManager/conf.d
echo -e "[keyfile]\nunmanaged-devices=interface-name:wgexpressvpn*" | sudo tee /etc/NetworkManager/conf.d/wgexpressvpn.conf > /dev/null

# Create a symlink for expressvpnctl (skip if already exists, update only)
sudo ln -s /opt/expressvpn/bin/expressvpnctl /usr/local/bin/expressvpnctl

# Copying the service file
sudo cp /tmp/expressvpn/x64/installfiles/expressvpn-service.service /etc/systemd/system/expressvpn.service

# Restart systemd and activate the service
sudo systemctl daemon-reload
sudo systemctl enable expressvpn
sudo systemctl start expressvpn

# system reboot and final check
(sudo systemctl reboot)
